O(log n) - Logarithmic Time

Example 1 Codes
namespace BigONotationLab;

public partial class Form1 : Form
{
    public Form1()
    {
        InitializeComponent();
    }

    private void btnGetFirst_Click(object sender, EventArgs e)
    {



        int[] numbers = { 10, 20, 30, 40, 50, 60, 70, 80 };

        int target = 40;
        int left = 0;
        int right = numbers.Length - 1;
        int position = -1;

        while (left <= right)
        {
            int middle = (left + right) / 2;

            if (numbers[middle] == target)
            {
                position = middle;
                break;
            }
            else if (numbers[middle] < target)
            {
                left = middle + 1;
            }
            else
            {
                right = middle - 1;
            }
        }

        if (position != -1)
        {
            lblResult.Text = "40 found at index: " + position;
        }
        else
        {
            lblResult.Text = "Number not found.";
        }
    }
}

    public partial class Form2 : Form
    {
        public Form2()
        {
            InitializeComponent();
        }

        private void btnDisplay_Click(object sender, EventArgs e)
        {
            int[] numbers = { 14, 27, 39, 42, 56, 68 };

            if (!int.TryParse(txtTarget.Text, out int target))
            {
                lblResult.Text = "Please enter a valid number.";
                return;
            }

            int position = -1;
            int checks = 0;

            foreach (int number in numbers)
            {
                checks++;

                if (number == target)
                {
                    position = checks - 1;
                    break;
                }
            }

            if (position != -1)
            {
                lblResult.Text = "Found at index: " + position
                    + " | Numbers checked: " + checks;
            }
            else
            {
                lblResult.Text = "Number not found | Numbers checked: " + checks;
            }
        }
    }
}

            string result = "";

            for (int i = 0; i < numbers.Length; i++)
            {
                for (int j = i + 1; j < numbers.Length; j++)
                {
                    if (numbers[i] == numbers[j])
                    {
                        result += "Duplicate: " + numbers[i]
                            + Environment.NewLine;
                    }
                }
            }

            if (result == "")
            {
                result = "No duplicates found.";
            }

            lblResult.Text = result;

        }
    }

Example 1 Codes
namespace BigONotationLab
{
    public partial class Form2 : Form
    {
        public Form2()
        {
            InitializeComponent();
        }

        private void btnDisplay_Click(object sender, EventArgs e)
        {


            int Fibonacci(int n)
            {
                if (n <= 1)
                {
                    return n;
                }

                return Fibonacci(n - 1) + Fibonacci(n - 2);
            }

            int number = 6;
            int result = Fibonacci(number);

            lblResult.Text = "Fibonacci(" + number + ") = " + result;

        }
    }
}

namespace BigONotationLab
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

       
        

        private void btnGenerate_Click(object sender, EventArgs e)
        {
            string output = "";
            GenerateStrings(3, "", ref output);
            lblResult.Text = output;
        }

        private void GenerateStrings(int n, string current, ref string result)
        {
            if (n == 0)
            {
                result += current + Environment.NewLine;
                return;
            }

            GenerateStrings(n - 1, current + "0", ref result);
            GenerateStrings(n - 1, current + "1", ref result);
        }
    }
}
